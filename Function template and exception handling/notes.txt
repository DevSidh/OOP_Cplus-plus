===============================================================================
                    C++ FUNCTION TEMPLATES AND EXCEPTION HANDLING
===============================================================================

1. FUNCTION TEMPLATES
===============================================================================

1.1 BASIC FUNCTION TEMPLATES
----------------------------
Function templates allow you to write generic functions that work with any data type.

Syntax:
    template <typename T>
    return_type function_name(parameters) {
        // function body
    }

Example:
    template <typename T>
    T add(T a, T b) {
        return a + b;
    }

    int main() {
        cout << add(5, 3);           // T = int, output: 8
        cout << add(2.5, 3.5);       // T = double, output: 6
        cout << add("Hello", " World"); // T = string, output: Hello World
        return 0;
    }

1.2 MULTIPLE TEMPLATE PARAMETERS
---------------------------------
You can have multiple template parameters.

Example:
    template <typename T, typename U>
    void print_pair(T first, U second) {
        cout << first << " " << second << endl;
    }

    int main() {
        print_pair(10, "hello");     // T=int, U=string
        print_pair(3.14, 42);        // T=double, U=int
        return 0;
    }

1.3 DEFAULT TEMPLATE ARGUMENTS
-------------------------------
You can provide default types for template parameters.

Example:
    template <typename T = int>
    T multiply(T a, T b) {
        return a * b;
    }

    int main() {
        cout << multiply(5, 3);      // Uses default int
        cout << multiply(2.5, 3.5);  // Explicitly uses double
        return 0;
    }

1.4 NON-TYPE TEMPLATE PARAMETERS
---------------------------------
Templates can also accept constant values as parameters.

Example:
    template <typename T, int size>
    void fill_array(T arr[]) {
        for (int i = 0; i < size; i++) {
            arr[i] = i * 2;
        }
    }

    int main() {
        int arr[5];
        fill_array<int, 5>(arr);
        for (int val : arr) cout << val << " ";  // Output: 0 2 4 6 8
        return 0;
    }

1.5 FUNCTION TEMPLATE SPECIALIZATION
-------------------------------------
You can create specialized versions of templates for specific types.

Example:
    template <typename T>
    void print_type(T value) {
        cout << "Generic template" << endl;
    }

    // Specialization for int
    template <>
    void print_type<int>(int value) {
        cout << "Specialized for int: " << value << endl;
    }

    int main() {
        print_type(3.14);      // Generic template
        print_type(42);        // Specialized for int
        return 0;
    }

1.6 VARIADIC TEMPLATES (C++11)
------------------------------
Templates that can accept variable number of arguments.

Example:
    template <typename T>
    void print_args(T single) {
        cout << single << endl;
    }

    template <typename T, typename... Args>
    void print_args(T first, Args... rest) {
        cout << first << " ";
        print_args(rest...);
    }

    int main() {
        print_args(1, 2, 3, 4, 5);  // Output: 1 2 3 4 5
        return 0;
    }

===============================================================================
2. EXCEPTION HANDLING
===============================================================================

2.1 BASIC TRY-CATCH BLOCKS
---------------------------
Used to catch and handle runtime errors.

Syntax:
    try {
        // Code that may throw an exception
    }
    catch (exception_type e) {
        // Handle the exception
    }

Example:
    int main() {
        try {
            int arr[5] = {1, 2, 3, 4, 5};
            cout << arr[10];  // Out of bounds
        }
        catch (exception &e) {
            cout << "Exception caught!" << endl;
        }
        return 0;
    }

2.2 THROWING EXCEPTIONS
------------------------
Use throw to create and throw exceptions.

Example:
    void divide(int a, int b) {
        if (b == 0) {
            throw runtime_error("Division by zero!");
        }
        cout << "Result: " << (a / b) << endl;
    }

    int main() {
        try {
            divide(10, 0);
        }
        catch (runtime_error &e) {
            cout << "Error: " << e.what() << endl;  // what() returns error message
        }
        return 0;
    }

2.3 MULTIPLE CATCH BLOCKS
--------------------------
Handle different types of exceptions.

Example:
    int main() {
        try {
            int choice;
            cin >> choice;
            
            if (choice == 1) throw invalid_argument("Invalid argument!");
            if (choice == 2) throw out_of_range("Out of range!");
            if (choice == 3) throw runtime_error("Runtime error!");
        }
        catch (invalid_argument &e) {
            cout << "Caught invalid argument: " << e.what() << endl;
        }
        catch (out_of_range &e) {
            cout << "Caught out of range: " << e.what() << endl;
        }
        catch (runtime_error &e) {
            cout << "Caught runtime error: " << e.what() << endl;
        }
        catch (exception &e) {
            cout << "Generic exception caught" << endl;
        }
        return 0;
    }

2.4 CUSTOM EXCEPTIONS
---------------------
Create your own exception classes.

Example:
    class MyException : public exception {
        string message;
    public:
        MyException(string msg) : message(msg) {}
        
        const char* what() const noexcept override {
            return message.c_str();
        }
    };

    int main() {
        try {
            throw MyException("Custom error message");
        }
        catch (MyException &e) {
            cout << "Custom error: " << e.what() << endl;
        }
        return 0;
    }

2.5 CATCH-ALL HANDLER
---------------------
Catch all exceptions using (...).

Example:
    int main() {
        try {
            throw 42;  // throw int
        }
        catch (...) {  // Catches any exception
            cout << "Unknown exception caught!" << endl;
        }
        return 0;
    }

2.6 NESTED TRY-CATCH
--------------------
Try-catch blocks can be nested.

Example:
    int main() {
        try {
            try {
                throw invalid_argument("Inner exception");
            }
            catch (logic_error &e) {
                cout << "Inner catch: " << e.what() << endl;
            }
            throw out_of_range("Outer exception");
        }
        catch (out_of_range &e) {
            cout << "Outer catch: " << e.what() << endl;
        }
        return 0;
    }

2.7 FINALLY-LIKE BEHAVIOR (RAII)
---------------------------------
C++ doesn't have finally, but uses RAII (Resource Acquisition Is Initialization).

Example:
    class ResourceManager {
    public:
        ResourceManager() { cout << "Resource acquired" << endl; }
        ~ResourceManager() { cout << "Resource released" << endl; }
    };

    int main() {
        try {
            ResourceManager rm;
            throw runtime_error("Something went wrong");
        }
        catch (exception &e) {
            cout << "Caught: " << e.what() << endl;
        }
        // Destructor called automatically
        return 0;
    }

2.8 NOEXCEPT SPECIFICATION (C++11)
-----------------------------------
Specify that a function doesn't throw exceptions.

Example:
    void safe_function() noexcept {
        cout << "This function doesn't throw" << endl;
    }

    void may_throw() noexcept(false) {
        throw runtime_error("May throw");
    }

    int main() {
        safe_function();
        return 0;
    }

2.9 COMMON STANDARD EXCEPTIONS
-------------------------------
exception              - Base class for all exceptions
logic_error            - Runtime logic error
invalid_argument       - Invalid argument passed
out_of_range           - Value outside expected range
runtime_error          - Generic runtime error
bad_alloc              - Memory allocation failure
bad_cast               - Invalid type cast
bad_exception          - Unexpected exception

Example:
    #include <stdexcept>

    int main() {
        try {
            vector<int> vec = {1, 2, 3};
            cout << vec.at(10);  // Throws out_of_range
        }
        catch (out_of_range &e) {
            cout << "Error: " << e.what() << endl;
        }
        return 0;
    }

===============================================================================
3. COMBINING TEMPLATES AND EXCEPTIONS
===============================================================================

3.1 TEMPLATE FUNCTIONS WITH EXCEPTION HANDLING
-----------------------------------------------

Example:
    template <typename T>
    T safe_divide(T a, T b) {
        if (b == 0) {
            throw invalid_argument("Divisor cannot be zero");
        }
        return a / b;
    }

    int main() {
        try {
            cout << safe_divide(10, 2);    // Output: 5
            cout << safe_divide(10, 0);    // Throws exception
        }
        catch (invalid_argument &e) {
            cout << "Error: " << e.what() << endl;
        }
        return 0;
    }

3.2 TEMPLATE CLASS WITH EXCEPTION HANDLING
--------------------------------------------

Example:
    template <typename T>
    class Stack {
        vector<T> elements;
    public:
        void push(T value) {
            elements.push_back(value);
        }

        T pop() {
            if (elements.empty()) {
                throw underflow_error("Stack is empty!");
            }
            T value = elements.back();
            elements.pop_back();
            return value;
        }

        bool is_empty() const {
            return elements.empty();
        }
    };

    int main() {
        Stack<int> stk;
        try {
            stk.push(10);
            stk.push(20);
            cout << stk.pop() << endl;  // Output: 20
            cout << stk.pop() << endl;  // Output: 10
            cout << stk.pop() << endl;  // Throws exception
        }
        catch (underflow_error &e) {
            cout << "Error: " << e.what() << endl;
        }
        return 0;
    }

===============================================================================
4. BEST PRACTICES
===============================================================================

1. Use specific exception types rather than generic exception
2. Provide meaningful error messages
3. Avoid throwing exceptions in destructors
4. Use RAII for resource management
5. Specify noexcept when functions don't throw
6. Catch exceptions by const reference
7. Don't catch (...) unless you need to
8. Use templates for generic programming
9. Specialize templates only when necessary
10. Keep exception hierarchies simple and logical
